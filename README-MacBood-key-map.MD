# MacBook key map

-- Go back to previous location
control + '-'

-- comment block
shift + command + 7

-- clear terminal
command + k

-- To do reverse-search ctrl + R and type any clue you have and that will return your previous commands matching the words you type.
control + r // bck-i-search -> type search string e.g. --dev
